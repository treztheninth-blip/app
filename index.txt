      // Angle parameter t goes from 0 to 1 clockwise starting at bottom-left of Medallion and ending at bottom-right of Medallion.
      // To create authentic visual separation between decades, each decade has:
      // - 1 Large Pater bead (isolated with chain spacing before & after)
      // - 10 Small Ave beads (clustered smoothly)
      // - 1 Golden Doxology clasp before the next Pater bead
      const loopUnits = [];
      let cumulativeArc = 0;

      for (let d = 1; d <= 5; d++) {
        // Gap before Large Pater Bead
        cumulativeArc += (d === 1) ? 1.55 : 1.45;
        loopUnits.push({
          decade: d,
          nodeType: 'pater',
          beadInDecade: 0,
          prayerId: 'our_father',
          shortLabel: `Decade ${d} • Mystery & Our Father`,
          intention: `Announce Mystery ${d} & Pray Our Father`,
          arcPos: cumulativeArc
        });

        // Gap after Large Pater Bead before 1st Ave
        cumulativeArc += 1.55;

        for (let a = 1; a <= 10; a++) {
          if (a > 1) cumulativeArc += 0.92;
          loopUnits.push({
            decade: d,
            nodeType: 'ave',
            beadInDecade: a,
            prayerId: 'hail_mary',
            shortLabel: `Decade ${d} • Hail Mary ${a}/10`,
            intention: `Contemplating Mystery ${d} (Bead ${a} of 10)`,
            arcPos: cumulativeArc
          });
        }

        // Doxology clasp on the chain after 10th Ave bead
        cumulativeArc += 1.25;
        loopUnits.push({
          decade: d,
          nodeType: 'doxology',
          beadInDecade: 11,
          prayerId: 'decade_doxology',
          shortLabel: `Decade ${d} • Glory Be & Fatima Prayer`,
          intention: `Concluding Decade ${d} with Glory Be & Fatima Prayer`,
          arcPos: cumulativeArc
        });
      }

      const totalArcSpan = cumulativeArc + 1.45;

      // Helper to convert normalized parameter u in (0, 1) to teardrop-ellipse (x, y)
      // u = 0 is bottom center (220, 355), moving clockwise:
      // left side (u < 0.5), top center (u = 0.5), right side (u > 0.5)
      function getLoopPoint(u) {
        // Angle theta: at u=0, bottom (pi/2). Moving clockwise in screen coords (y down):
        // theta goes from pi/2 -> pi (left) -> 3pi/2 (top) -> 2pi (right) -> 5pi/2 (bottom)
        const theta = (Math.PI / 2) + u * (2 * Math.PI);
        const cx = 220;
        const cy = 190;
        const rx = 166;
        const ry = 158;

        // Teardrop pinch factor near bottom (sin(theta) near 1)
        const bottomFactor = Math.max(0, Math.sin(theta));
        const pinchX = 1 - 0.16 * Math.pow(bottomFactor, 2.2);

        const x = cx + rx * pinchX * Math.cos(theta);
        const y = cy + ry * Math.sin(theta) + (8 * Math.pow(bottomFactor, 3));
        return { x: Number(x.toFixed(1)), y: Number(y.toFixed(1)) };
      }

      loopUnits.forEach((unit, idx) => {
        const u = unit.arcPos / totalArcSpan;
        const pt = getLoopPoint(u);
        ROSARY_STEPS.push({
          stepIndex: 6 + idx,
          nodeType: unit.nodeType,
          section: unit.decade,
          decade: unit.decade,
          beadInDecade: unit.beadInDecade,
          prayerId: unit.prayerId,
          shortLabel: unit.shortLabel,
          intention: unit.intention,
          x: pt.x,
          y: pt.y
        });
      });

      // Final Step (66): Centerpiece Medallion (Hail Holy Queen & Concluding Prayer)
      ROSARY_STEPS.push({
        stepIndex: 66,
        nodeType: 'medallion',
        section: 'closing',
        decade: 0,
        beadInDecade: 0,
        prayerId: 'hail_holy_queen',
        shortLabel: 'Centerpiece Medallion • Salve Regina',
        intention: 'Concluding Marian Anthem & Rosary Prayer',
        x: 220,
        y: 353
      });
    }

    /*************************************************************************
     * 5. WEB AUDIO API SYNTHESIZER (WOODEN BEAD CLICK & SINGING BELL CHIME)
     *************************************************************************/
    let audioCtx = null;

    function getAudioContext() {
      if (!audioCtx) {
        const AudioContextClass = window.AudioContext || window.webkitAudioContext;
        if (AudioContextClass) {
          audioCtx = new AudioContextClass();
        }
      }
      if (audioCtx && audioCtx.state === 'suspended') {
        audioCtx.resume();
      }
      return audioCtx;
    }

    function playRosarySound(nodeType) {
      if (!state.audioEnabled) return;
      try {
        const ctx = getAudioContext();
        if (!ctx) return;
        const now = ctx.currentTime;

        if (nodeType === 'pater' || nodeType === 'medallion' || nodeType === 'crucifix') {
          // Warm Liturgical Bell / Singing Bowl Chime
          const baseFreq = nodeType === 'medallion' ? 528 : 440;
          const harmonics = [1, 2.0, 2.756];
          const gains = [0.11, 0.045, 0.02];

          harmonics.forEach((mult, i) => {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(baseFreq * mult, now);

            gain.gain.setValueAtTime(0.0001, now);
            gain.gain.linearRampToValueAtTime(gains[i], now + 0.015);
            gain.gain.exponentialRampToValueAtTime(0.0001, now + (1.3 - i * 0.3));

            osc.connect(gain);
            gain.connect(ctx.destination);
            osc.start(now);
            osc.stop(now + 1.35);
          });
        } else if (nodeType === 'doxology') {
          // Subtle Golden Clasp Shimmer
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(660, now);
          osc.frequency.exponentialRampToValueAtTime(880, now + 0.12);

          gain.gain.setValueAtTime(0.001, now);
          gain.gain.linearRampToValueAtTime(0.06, now + 0.01);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.32);

          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start(now);
          osc.stop(now + 0.35);
        } else {
          // Tactile Olive-Wood Bead Click
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.type = 'sine';
          osc.frequency.setValueAtTime(290, now);
          osc.frequency.exponentialRampToValueAtTime(115, now + 0.045);

          gain.gain.setValueAtTime(0.001, now);
          gain.gain.linearRampToValueAtTime(0.14, now + 0.004);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.055);

          osc.connect(gain);
          gain.connect(ctx.destination);
          osc.start(now);
          osc.stop(now + 0.06);
        }
      } catch (e) {
        // Ignore audio errors if blocked by browser autoplay policy
      }
    }

    function triggerHaptic(nodeType) {
      if (!state.hapticsEnabled) return;
      try {
        if ('vibrate' in navigator) {
          if (nodeType === 'pater' || nodeType === 'medallion') {
            navigator.vibrate([22, 35, 22]);
          } else {
            navigator.vibrate(12);
          }
        }
      } catch (e) {
        // Ignore unsupported vibration API
      }
    }

    function detectTodaysMystery() {
      const dayIndex = new Date().getDay(); // 0 = Sun, 1 = Mon, ..., 6 = Sat
      const dayNames = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];
      const dayToMysteryMap = {
        0: 'glorious',  // Sunday
        1: 'joyful',    // Monday
        2: 'sorrowful', // Tuesday
        3: 'glorious',  // Wednesday
        4: 'luminous',  // Thursday
        5: 'sorrowful', // Friday
        6: 'joyful'     // Saturday
      };

      state.todaysDayName = dayNames[dayIndex];
      state.todaysMysteryId = dayToMysteryMap[dayIndex];
      state.mysterySetId = state.todaysMysteryId;
      state.isManualMysteryOverride = false;
    }

    function selectMysterySet(mysteryId, isManual = true) {
      if (!MYSTERY_SETS[mysteryId]) return;

      state.isManualMysteryOverride = isManual && (mysteryId !== state.todaysMysteryId);
      updateMysteryHeaderUI();
      renderCurrentStepUI();
      renderSvgRosaryState();
    }

    function renderSvgRosaryState() {
      const current = ROSARY_STEPS[state.currentStep];
      const halo = document.getElementById('svgActiveHalo');
      if (halo && current) {
        halo.setAttribute('cx', current.x);
        halo.setAttribute('cy', current.y + (current.nodeType === 'crucifix' ? 2 : 0));
      }

      const isDark = document.documentElement.classList.contains('dark');
      const unvisitedFill = isDark ? 'url(#beadUnvisitedDark)' : 'url(#beadUnvisitedLight)';

      ROSARY_STEPS.forEach(step => {
        const g = document.getElementById(`rosary-node-${step.stepIndex}`);
        const shape = document.getElementById(`node-shape-${step.stepIndex}`);
        if (!g || !shape) return;

        const isActive = step.stepIndex === state.currentStep;
        const isCompleted = step.stepIndex < state.currentStep;

        g.classList.toggle('rosary-node-active', isActive);

        if (isActive) {
          shape.setAttribute('fill', 'url(#beadActiveGrad)');
          shape.setAttribute('stroke', '#B45309');
          shape.setAttribute('filter', 'url(#softGlow)');
        } else if (isCompleted) {
          shape.setAttribute('fill', 'url(#beadCompletedGrad)');
          shape.setAttribute('stroke', '#92400E');
          shape.removeAttribute('filter');
        } else {
          if (step.nodeType === 'pater') {
            shape.setAttribute('fill', 'url(#paterBeadGrad)');
            shape.setAttribute('stroke', isDark ? '#60A5FA' : '#1E3A8A');
          } else if (step.nodeType === 'crucifix' || step.nodeType === 'medallion') {
            shape.setAttribute('fill', 'url(#beadCompletedGrad)');
            shape.setAttribute('stroke', '#78350F');
          } else if (step.nodeType === 'doxology') {
            shape.setAttribute('fill', isDark ? '#334155' : '#E7E0D0');
            shape.setAttribute('stroke', '#B48327');
          } else {
            shape.setAttribute('fill', unvisitedFill);
            shape.setAttribute('stroke', isDark ? '#475569' : '#A8A29E');
          }
          shape.removeAttribute('filter');
        }
      });

      // Update Center Emblem Text inside the SVG Loop
      const activeSet = MYSTERY_SETS[state.mysterySetId];
      document.getElementById('svgCenterMysteryTitle').textContent = activeSet.name;
      document.getElementById('svgCenterLatinTitle').textContent = activeSet.latinName;

      const centerDecadeStatus = document.getElementById('svgCenterDecadeStatus');
      const centerSubStatus = document.getElementById('svgCenterSubStatus');

      if (current.section === 'intro') {
        centerDecadeStatus.textContent = 'Introductory Rites';
        centerSubStatus.textContent = current.shortLabel;
      } else if (current.section === 'closing') {
        centerDecadeStatus.textContent = 'Closing Anthem';
        centerSubStatus.textContent = 'Salve Regina';
      } else {
        const m = activeSet.mysteries[current.decade - 1];
        centerDecadeStatus.textContent = `Decade ${current.decade} • ${m.title}`;
        centerSubStatus.textContent = current.beadInDecade >= 1 && current.beadInDecade <= 10
          ? `Hail Mary ${current.beadInDecade} of 10`
          : current.shortLabel;
      }
    }

            ? `Hail Mary ${step.beadInDecade} of 10`
            : step.nodeType === 'pater'
              ? 'Decade Pater Bead (Our Father)'
              : 'Decade Doxology (Glory Be & Fatima Prayer)';

        const dotsContainer = document.getElementById('decadeMiniDots');
        dotsContainer.innerHTML = '';
        for (let b = 1; b <= 10; b++) {
          const dot = document.createElement('span');
          const isCurrentAve = step.beadInDecade === b;
          const isPastAve = step.beadInDecade > b;
          dot.className = isCurrentAve
            ? 'w-3 h-3 rounded-full bg-sacred-gold ring-2 ring-sacred-gold/40 transition-all'
            : isPastAve
              ? 'w-2.5 h-2.5 rounded-full bg-sacred-gold/75 transition-all'
              : 'w-2.5 h-2.5 rounded-full bg-parchment-300 dark:bg-vigil-700 transition-all';
          dotsContainer.appendChild(dot);
        }
      }

      // 3. Update Active Prayer Card
      document.getElementById('prayerStepBadge').textContent = step.shortLabel;
      document.getElementById('prayerIntentionBadge').textContent = step.intention;
      document.getElementById('activePrayerTitle').textContent = prayerObj.title;
      document.getElementById('activePrayerLatinTitle').textContent = prayerObj.latinTitle;
      document.getElementById('activePrayerBody').textContent = getPrayerText(step.prayerId);
      document.getElementById('activePrayerLatinBox').textContent = prayerObj.latinIncipit;

      // Trigger subtle fade animation on prayer card content
      const animBox = document.getElementById('prayerContentAnimated');
      animBox.classList.remove('animate-prayer-transition');
      void animBox.offsetWidth; // Reflow
      animBox.classList.add('animate-prayer-transition');

      // Custom status indicator
      const isCustomized = Boolean(state.customPrayers[step.prayerId]);
      const statusEl = document.getElementById('customTextStatusIndicator');
      statusEl.innerHTML = isCustomized
        ? `<span class="w-2 h-2 rounded-full bg-emerald-500"></span><span>Displaying your customized prayer text from browser storage.</span>`
        : `<span class="w-2 h-2 rounded-full bg-sacred-gold/70"></span><span>Modular Incipit Mode &mdash; Click "Prayer Texts" to customize full wording in your browser.</span>`;

      // 4. Update Navigation Buttons & Overall Progress Bar
      document.getElementById('prevBeadBtn').disabled = state.currentStep === 0;
      const nextBtnText = document.getElementById('nextBeadBtnText');
      nextBtnText.textContent = state.currentStep === ROSARY_STEPS.length - 1 ? 'Complete & Restart' : 'Next Bead';

      const pct = Math.round(((state.currentStep + 1) / ROSARY_STEPS.length) * 100);
      document.getElementById('progressPercentLabel').textContent = `${pct}%`;
      document.getElementById('overallProgressBar').style.width = `${pct}%`;
      document.getElementById('progressSectionName').textContent =
        step.section === 'intro'
          ? 'Introductory Rites'
          : step.section === 'closing'
            ? 'Closing Prayers'
            : `Decade ${step.decade} of 5`;

      // 5. Highlight Active Decade Jump Pill
      ['intro', 1, 2, 3, 4, 5, 'closing'].forEach(sec => {
        const btn = document.getElementById(`jumpBtn-${sec}`);
        if (!btn) return;
        if (step.section === sec) {
          btn.className = 'decade-jump-btn flex-1 py-1.5 px-2 rounded-lg text-[11px] font-mono font-bold bg-sacred-gold text-white border border-sacred-gold shadow-xs transition-all text-center';
        } else {
          btn.className = 'decade-jump-btn flex-1 py-1.5 px-2 rounded-lg text-[11px] font-mono font-medium bg-parchment-50 dark:bg-vigil-900 text-stone-600 dark:text-stone-300 border border-parchment-200 dark:border-vigil-700 hover:border-sacred-gold transition-all text-center';
        }
      });
    }

    /*************************************************************************
     * 9. NAVIGATION ACTIONS & KEYBOARD SHORTCUTS
     *************************************************************************/
    function goToStep(stepIndex, playFeedback = true) {
      if (stepIndex < 0 || stepIndex >= ROSARY_STEPS.length) return;
      state.currentStep = stepIndex;
      const current = ROSARY_STEPS[state.currentStep];

      if (playFeedback) {
        playRosarySound(current.nodeType);
        triggerHaptic(current.nodeType);
      }

      renderCurrentStepUI();
      renderSvgRosaryState();
    }

    function stepBead(delta) {
      if (state.currentStep === ROSARY_STEPS.length - 1 && delta > 0) {
        // Completed the Rosary -> return to Crucifix with celebratory chime
        goToStep(0, true);
        showToast('Rosary completed — Deo Gratias! Returned to Crucifix.');
        return;
      }
      const target = Math.max(0, Math.min(ROSARY_STEPS.length - 1, state.currentStep + delta));
      goToStep(target, true);
    }

    function jumpToSection(section) {
      const idx = ROSARY_STEPS.findIndex(s => s.section === section);
      if (idx !== -1) {
        goToStep(idx, true);
      }
    }

    function resetRosaryProgress() {
      goToStep(0, true);
      showToast('Returned to the Crucifix.');
    }

    function toggleLatinDisplay() {
      state.showLatin = !state.showLatin;
      const box = document.getElementById('activePrayerLatinBox');
      const btn = document.getElementById('latinToggleBtn');
      box.classList.toggle('hidden', !state.showLatin);
      btn.textContent = state.showLatin ? 'Hide Latin Incipit' : 'Show Latin Incipit';
    }

    /*************************************************************************
     * 10. PRAYER TEXT EDITOR MODAL & LOCALSTORAGE PERSISTENCE
     *************************************************************************/
    function loadStoredData() {
      try {
        const storedPrayers = localStorage.getItem(STORAGE_KEY_PRAYERS);
        if (storedPrayers) {
          state.customPrayers = JSON.parse(storedPrayers) || {};
        }
      } catch (e) {
        state.customPrayers = {};
      }

      try {
        const storedSettings = localStorage.getItem(STORAGE_KEY_SETTINGS);
        if (storedSettings) {
          const parsed = JSON.parse(storedSettings);
          if (typeof parsed.audioEnabled === 'boolean') state.audioEnabled = parsed.audioEnabled;
          if (typeof parsed.hapticsEnabled === 'boolean') state.hapticsEnabled = parsed.hapticsEnabled;
          if (typeof parsed.darkMode === 'boolean') state.darkMode = parsed.darkMode;
        }
      } catch (e) {
        // Use defaults
      }
    }

    function saveSettings() {
      try {
        localStorage.setItem(STORAGE_KEY_SETTINGS, JSON.stringify({
          audioEnabled: state.audioEnabled,
          hapticsEnabled: state.hapticsEnabled,
          darkMode: state.darkMode
        }));
      } catch (e) {
        // Ignore storage quota errors
      }
    }

    function populateEditorDropdown(selectedPrayerId) {
      const select = document.getElementById('editorPrayerSelect');
      select.innerHTML = '';
      Object.values(DEFAULT_PRAYER_LIBRARY).forEach(prayer => {
        const opt = document.createElement('option');
        opt.value = prayer.id;
        const isCustom = Boolean(state.customPrayers[prayer.id]);
        opt.textContent = `${prayer.title}${isCustom ? ' (Customized)' : ''}`;
        if (prayer.id === selectedPrayerId) opt.selected = true;
        select.appendChild(opt);
      });
      handleEditorPrayerChange();
    }

    function handleEditorPrayerChange() {
      const prayerId = document.getElementById('editorPrayerSelect').value;
      const prayer = DEFAULT_PRAYER_LIBRARY[prayerId];
      if (!prayer) return;

      document.getElementById('editorMetaTitle').textContent = prayer.title;
      document.getElementById('editorMetaLatin').textContent = prayer.latinTitle;
      document.getElementById('editorPrayerTextarea').value = getPrayerText(prayerId);
    }

    function openPrayerEditorModal(prayerId) {
      const targetId = prayerId || ROSARY_STEPS[state.currentStep].prayerId;
      populateEditorDropdown(targetId);
      const modal = document.getElementById('prayerEditorModal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function openPrayerEditorForCurrent() {
      const currentPrayerId = ROSARY_STEPS[state.currentStep].prayerId;
      openPrayerEditorModal(currentPrayerId);
    }

    function closePrayerEditorModal() {
      const modal = document.getElementById('prayerEditorModal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    function savePrayerFromEditor() {
      const prayerId = document.getElementById('editorPrayerSelect').value;
      const newText = document.getElementById('editorPrayerTextarea').value.trim();

      if (newText === '' || newText === DEFAULT_PRAYER_LIBRARY[prayerId].defaultText) {
        delete state.customPrayers[prayerId];
      } else {
        state.customPrayers[prayerId] = newText;
      }

      try {
        localStorage.setItem(STORAGE_KEY_PRAYERS, JSON.stringify(state.customPrayers));
      } catch (e) {
        // Ignore storage error
      }

      populateEditorDropdown(prayerId);
      renderCurrentStepUI();
      closePrayerEditorModal();
      showToast('Prayer text saved to localStorage.');
    }

    function resetCurrentPrayerToDefault() {
      const prayerId = document.getElementById('editorPrayerSelect').value;
      delete state.customPrayers[prayerId];
      try {
        localStorage.setItem(STORAGE_KEY_PRAYERS, JSON.stringify(state.customPrayers));
      } catch (e) {}
      document.getElementById('editorPrayerTextarea').value = DEFAULT_PRAYER_LIBRARY[prayerId].defaultText;
      populateEditorDropdown(prayerId);
      renderCurrentStepUI();
      showToast('Restored default modular incipit.');
    }

    function resetAllPrayersToDefault() {
      state.customPrayers = {};
      try {
        localStorage.removeItem(STORAGE_KEY_PRAYERS);
      } catch (e) {}
      const prayerId = document.getElementById('editorPrayerSelect').value;
      populateEditorDropdown(prayerId);
      renderCurrentStepUI();
      showToast('All prayers reset to default incipits.');
    }

    /*************************************************************************
     * 11. AUDIO, HAPTICS, THEME & TOOLTIP HELPERS
     *************************************************************************/
    function toggleAudio() {
      state.audioEnabled = !state.audioEnabled;
      saveSettings();
      updateUtilityButtonsUI();
      if (state.audioEnabled) playRosarySound('ave');
      showToast(state.audioEnabled ? 'Synthesized Rosary audio enabled' : 'Audio muted');
    }

    function toggleHaptics() {
      state.hapticsEnabled = !state.hapticsEnabled;
      saveSettings();
      updateUtilityButtonsUI();
      if (state.hapticsEnabled) triggerHaptic('ave');
      showToast(state.hapticsEnabled ? 'Tactile haptics enabled' : 'Haptics disabled');
    }

    function toggleTheme() {
      state.darkMode = !state.darkMode;
      applyTheme();
      saveSettings();
      renderSvgRosaryState();
    }

    function applyTheme() {
      const htmlEl = document.documentElement;
      if (state.darkMode) {
        htmlEl.classList.add('dark');
        htmlEl.classList.remove('light');
      } else {
        htmlEl.classList.remove('dark');
        htmlEl.classList.add('light');
      }
      document.getElementById('themeIconSun').classList.toggle('hidden', !state.darkMode);
      document.getElementById('themeIconMoon').classList.toggle('hidden', state.darkMode);
    }

    function updateUtilityButtonsUI() {
      document.getElementById('audioIconOn').classList.toggle('hidden', !state.audioEnabled);
      document.getElementById('audioIconOff').classList.toggle('hidden', state.audioEnabled);

      const hapticIcon = document.getElementById('hapticIcon');
      if (state.hapticsEnabled) {
        hapticIcon.classList.remove('text-stone-400');
        hapticIcon.classList.add('text-sacred-gold', 'dark:text-sacred-goldLight');
      } else {
        hapticIcon.classList.add('text-stone-400');
        hapticIcon.classList.remove('text-sacred-gold', 'dark:text-sacred-goldLight');
      }
    }

    function showBeadTooltip(e, text) {
      const tip = document.getElementById('beadTooltip');
      tip.textContent = text;
      tip.classList.remove('hidden');
      moveBeadTooltip(e);
    }

    function moveBeadTooltip(e) {
      const tip = document.getElementById('beadTooltip');
      tip.style.left = `${e.clientX + 14}px`;
      tip.style.top = `${e.clientY - 28}px`;
    }

    function hideBeadTooltip() {
      document.getElementById('beadTooltip').classList.add('hidden');
    }

    let toastTimer = null;
    function showToast(msg) {
      const banner = document.getElementById('toastBanner');
      document.getElementById('toastMessage').textContent = msg;
      banner.classList.remove('hidden');
      if (toastTimer) clearTimeout(toastTimer);
      toastTimer = setTimeout(() => {
        banner.classList.add('hidden');
      }, 2600);
    }

    /*************************************************************************
     * 12. INITIALIZATION & KEYBOARD LISTENERS
     *************************************************************************/
    window.addEventListener('keydown', (e) => {
      // Do not intercept keys if user is typing in the Prayer Editor Modal
      const modal = document.getElementById('prayerEditorModal');
      if (!modal.classList.contains('hidden')) {
        if (e.key === 'Escape') closePrayerEditorModal();
        return;
      }

      if (e.key === 'ArrowRight' || e.key === ' ') {
        e.preventDefault();
        stepBead(1);
      } else if (e.key === 'ArrowLeft') {
        e.preventDefault();
        stepBead(-1);
      }
    });

      loadStoredData();
      applyTheme();
      updateUtilityButtonsUI();
      buildRosarySteps();
      buildSvgRosaryElements();
      detectTodaysMystery();
      updateMysteryHeaderUI();
      renderCurrentStepUI();
      renderSvgRosaryState();
    });
